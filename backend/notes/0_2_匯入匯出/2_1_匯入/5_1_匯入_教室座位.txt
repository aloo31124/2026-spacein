src\services\roomToSite.service.js
    /* [空間管理] [教室座位] 匯入, 教室,座位, ownerId 綁定 */
    exports.importRoomSite = async (ownerId, roomToSiteList) => {
        try {
            /** 1. 檢查 教室名稱, 是否空值, 是否存在系統, 不存在便新增 教室 */
            return roomToSiteList.map(async roomToSite => {
                let message = "";
                let room = await roomService.findByName(roomToSite.roomName);
                if(!room) {
                    room = await roomService.add({name: roomToSite.roomName}, ownerId);
                    console.log(`[importRoomSite] [空間管理] [教室座位] 匯入, [教室]新增成功 newRoom : ${room.id, room.name}`);
                    message = message + ` ${roomToSite.roomName}:[教室]另外新增; `;
                } else {
                    console.log(`[importRoomSite] [空間管理] [教室座位] 匯入, [教室]已存在`);
                    message = message + ` ${roomToSite.roomName}:[教室]已存在; `;
                }
                /** 2. 檢查 座位名稱, 是否空值, 是否存在系統, 不存在便新增 座位 */
                const roomId = room.id;
                if(!roomId) {
                    console.log(`[importRoomSite]  [空間管理] [教室座位] 匯入, [教室]資訊異常 room : ${room}`);
                    message = message + ` ${roomToSite.siteName}:[教室] 資訊異常; `;
                    return await {...roomToSite, message};
                }
                const checkRoomToSite = await RoomToSite.findByName(roomToSite.siteName);
                if(checkRoomToSite && checkRoomToSite.name) {
                    console.log(`[importRoomSite]  [空間管理] [教室座位] 匯入, [教室座位]已存在 : ${checkRoomToSite.name}`);
                    message = message + ` ${roomToSite.siteName}:[教室座位] 已存在 : ${checkRoomToSite.name} `;
                    return await {...roomToSite, message};
                }

                const newRoomToSite = await RoomToSite.add({roomId, name: roomToSite.siteName});
                if(newRoomToSite) {
                    console.log(`[importRoomSite]  [空間管理] [教室座位] 匯入, [教室座位] 新增成功 newRoomToSite : ${newRoomToSite}`);
                    message = message + ` ${roomToSite.siteName}:[教室座位] 新增成功; `;
                } else {
                    console.log(`[importRoomSite]  [空間管理] [教室座位] 匯入, [教室座位] 新增失敗 newRoomToSite : ${newRoomToSite}`);
                    message = message + ` ${roomToSite.siteName}:[教室座位] 新增失敗; `;
                }
                return await {...roomToSite, message};
            });
        } catch (error) {
            throw new Error(`[importRoomSite] error: ${error.message}`);
        }
    }
