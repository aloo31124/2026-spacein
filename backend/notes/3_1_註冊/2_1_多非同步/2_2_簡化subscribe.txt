import { switchMap, tap, catchError, of } from 'rxjs';

const userId = this.googleAuthService.getCurrentUser().userFirestoreId;

this.ecpayService.getUserToPayment(userId).pipe(
  tap(res => {
    if (!res.result) {
      this.paymentDescription = "尚未購買，需選擇方案才可升級為擁有者";
      alert(this.paymentDescription);
      this.routeUrlRecordService.nextPage("pay-plan-list", {});
      throw new Error("尚未購買");
    }

    const { purchaseDate, expiryDate } = res.result;
    const currentDate = new Date();

    if (currentDate >= new Date(expiryDate)) {
      this.paymentDescription = "購買方案已過期，需續約喔";
      alert(this.paymentDescription);
      this.isExpired = true;
    } else {
      this.paymentDescription = "已購買方案，成為擁有者";
      alert(this.paymentDescription);
    }

    if (!res.result.paymentId) {
      alert("查無該擁有者方案資訊");
      throw new Error("paymentId 不存在");
    }
  }),
  switchMap(res => this.ecpayService.getPayment(res.result.paymentId)),
  catchError(err => {
    console.warn("流程中斷:", err.message);
    return of(null); // 中斷流程，但不進入 subscribe
  })
).subscribe(res => {
  if (res && res.result) {
    this.payment = res.result;
  }
});

import { switchMap } from 'rxjs/operators';

const userId = this.googleAuthService.getCurrentUser().userFirestoreId;

this.ecpayService.getUserToPayment(userId)
  .pipe(
    switchMap(paymentRes => {
      // 判斷是否已購買或過期
      if (!paymentRes.result) {
        this.paymentDescription = "尚未購買，需選擇方案才可升級為擁有者";
        alert(this.paymentDescription);
        this.routeUrlRecordService.nextPage("pay-plan-list", {});
        return throwError(() => new Error('尚未購買'));
      }

      const { purchaseDate, expiryDate } = paymentRes.result;
      const currentDate = new Date();
      
      if (currentDate >= new Date(expiryDate)) {
        this.paymentDescription = "購買方案已過期，需續約喔";
        alert(this.paymentDescription);
        this.isExpired = true;
      } else {
        this.paymentDescription = "已購買方案，成為擁有者";
        alert(this.paymentDescription);
      }

      // 取得方案資訊
      if (!paymentRes.result.paymentId) {
        alert("查無該擁有者方案資訊");
        return throwError(() => new Error('查無方案資訊'));
      }

      return this.ecpayService.getPayment(paymentRes.result.paymentId);
    })
  )
  .subscribe({
    next: paymentInfoRes => {
      this.payment = paymentInfoRes.result;
    },
    error: err => {
      console.error('處理過程中發生錯誤:', err.message);
      // 錯誤處理已在 switchMap 中完成
    }
  });
