------------------------------------------------

.img-btn {
    display: flex;
    justify-content: center;
    align-items: center;
    cursor: pointer;
    transition: transform 0.2s ease-in-out;
}

.img-btn:hover {
    animation: bounce 0.4s ease-in-out;
}

/* 輕微上下彈跳動畫 */
@keyframes bounce {
    0%   { transform: translateY(0); }
    50%  { transform: translateY(-5px); } /* 向上彈跳 */
    100% { transform: translateY(0); } /* 回到原位 */
}
