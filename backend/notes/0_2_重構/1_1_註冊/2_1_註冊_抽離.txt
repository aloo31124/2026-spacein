這裡的 `user.service.js` 仍然包含了部分 **資料庫查詢邏輯**，應將這些查詢行為提取到 `user.model.js`，
使 `service` 層專注於業務邏輯，而 `model` 層負責資料存取（CRUD）。  

## **調整後的架構**
- **`user.model.js`** 只負責 `Firestore` 操作，不涉及業務邏輯。
- **`user.service.js`** 只負責業務邏輯（如錯誤處理、記錄日誌等），並調用 `user.model.js` 來存取資料。

---

### **📌 調整後的 `user.model.js`**
```javascript
const db = require('../firestore');

class UserModel {
    static async findByAccount(account) {
        const snapshot = await db.collection('User').where('account', '==', account).get();
        return snapshot.empty ? null : { id: snapshot.docs[0].id, ...snapshot.docs[0].data() };
    }

    static async findByMail(mail) {
        const snapshot = await db.collection('User').where('mail', '==', mail).get();
        return snapshot.empty ? null : { id: snapshot.docs[0].id, ...snapshot.docs[0].data(), ref: snapshot.docs[0].ref };
    }

    static async create(user) {
        const docRef = await db.collection('User').add(user);
        const docSnapshot = await docRef.get();
        return { id: docRef.id, ...docSnapshot.data() };
    }

    static async update(userRef, fieldsToUpdate) {
        await userRef.update(fieldsToUpdate);
        const updatedSnapshot = await userRef.get();
        return { id: userRef.id, ...updatedSnapshot.data() };
    }
}

module.exports = UserModel;
```

### **📌 調整後的 `user.service.js`**
```javascript
const UserModel = require('../models/user.model');

const logger = (message, data) => {
    console.log(`[${new Date().toISOString()}] ${message}`, data);
};

const errorLogger = (message, error) => {
    console.error(`[${new Date().toISOString()}] ${message}`, error);
};

/* 取得使用者資訊, 依 account */
exports.getUserByAccount = async (account) => {
    logger("[getUserByAccount] 開始", { account });
    try {
        const user = await UserModel.findByAccount(account);
        if (!user) {
            logger("[getUserByAccount] 查無此帳號", { account });
            return null;
        }
        logger("[getUserByAccount] 成功", user);
        return user;
    } catch (error) {
        errorLogger("[getUserByAccount] 失敗", error);
        throw new Error("取得使用者資訊失敗");
    }
};

/* 新增使用者 */
exports.addUser = async (user) => {
    logger("[addUser] 開始", { mail: user.mail });
    try {
        const existingUser = await UserModel.findByMail(user.mail);
        if (existingUser) {
            logger("[addUser] 該信箱已註冊", { mail: user.mail });
            return null;
        }
        const newUser = await UserModel.create(user);
        logger("[addUser] 成功", newUser);
        return newUser;
    } catch (error) {
        errorLogger("[addUser] 失敗", error);
        throw new Error("新增使用者失敗");
    }
};

/* 更新使用者資訊 */
exports.updateUser = async (user) => {
    logger("[updateUser] 開始", { mail: user.mail });
    try {
        const existingUser = await UserModel.findByMail(user.mail);
        if (!existingUser) {
            logger("[updateUser] 查無此信箱使用者", { mail: user.mail });
            return null;
        }
        const { ref, ...userData } = existingUser;
        const updatedUser = await UserModel.update(ref, user);
        logger("[updateUser] 成功", updatedUser);
        return updatedUser;
    } catch (error) {
        errorLogger("[updateUser] 失敗", error);
        throw new Error("更新使用者資訊失敗");
    }
};

/* 登入 */
exports.login = async (account, password) => {
    logger("[login] 開始", { account });
    try {
        const user = await UserModel.findByAccount(account);
        if (!user) {
            logger("[login] 查無此帳號", { account });
            return false;
        }

        if (user.password !== password) {
            logger("[login] 密碼錯誤", { account });
            return false;
        }

        logger("[login] 成功", { account });
        return true;
    } catch (error) {
        errorLogger("[login] 失敗", error);
        throw new Error("登入失敗");
    }
};
```

### **📌 調整後的優勢**
| 層級      | 主要功能 | 是否與 Firestore 互動 | 是否處理業務邏輯 | 是否處理 HTTP Request |
|-----------|---------|----------------|----------------|----------------|
| **Model (`user.model.js`)**  | 負責 Firestore 資料存取 | ✅ 是 | ❌ 否 | ❌ 否 |
| **Service (`user.service.js`)** | 負責業務邏輯與錯誤處理 | ❌ 否（透過 Model 存取） | ✅ 是 | ❌ 否 |
| **Controller (`user.controller.js`)** | 處理 HTTP Request/Response | ❌ 否 | ❌ 主要調用 Service | ✅ 是 |

---

### **🔥 總結**
1. **Model 負責資料存取（CRUD）**，不包含業務邏輯，這樣 **資料操作方式更清晰**。
2. **Service 負責業務邏輯（錯誤處理、日誌、條件檢查）**，讓 **程式更可維護、更容易測試**。
3. **Controller 將保持簡潔，僅負責 HTTP Request 處理**，確保責任分離。

這樣的架構符合 **SOLID 原則（特別是單一職責原則 SRP）**，讓 Express 專案更模組化、可維護 🚀

在 node.js express 中, 請精要說明 
model, controller, service 主要功能與差異

------------------------------------------------

在 **Node.js + Express** 的架構中，
`Model`、`Controller` 和 `Service` 各自負責不同的業務邏輯，
通常遵循 **MVC（Model-View-Controller）模式**，
但 **Service 層** 是為了更好的架構解耦而額外增加的一層。以下是它們的 **主要功能與差異**：

---

### **1. Model（模型層）**
**主要功能：**
    - 負責與 **資料庫（Database）** 進行互動（CRUD 操作）。
    - 定義 **資料結構（Schema）** 及 **驗證規則**。
    - 一般使用 **ORM/ODM**（如 `Mongoose`、`Sequelize`）來處理資料庫操作。


---

### **2. Service（業務邏輯層）**
**主要功能：**
    - 負責 **業務邏輯處理**，確保程式的可讀性與模組化。
    - 讓 **Controller 與 Model 解耦**，避免 Controller 直接與資料庫互動。
    - 提供 **錯誤處理**、**資料轉換**、**額外邏輯計算**。


---

### **3. Controller（控制層）**
**主要功能：**
    - 負責 **接收 HTTP Request**，並調用 **Service 層** 來執行業務邏輯。
    - 處理 **請求/回應（Request/Response）**，如 `req.body`、`req.params`、`res.json()`。
    - **不應該** 直接與 `Model` 層互動，而是通過 `Service` 層來處理邏輯。

---

### **📌 差異對比**
| 層級       | 主要職責 | 是否與資料庫互動 | 是否包含業務邏輯 | 是否處理 HTTP 請求 |
|------------|---------|----------------|----------------|----------------|
| **Model**  | 資料庫操作（CRUD） | ✅ 是 | ❌ 否 | ❌ 否 |
| **Service** | 業務邏輯處理 | ✅ 可能會（透過 Model） | ✅ 是 | ❌ 否 |
| **Controller** | 處理 HTTP Request/Response | ❌ 否 | ❌ 主要調用 Service | ✅ 是 |

---

### **🔹 為何要有 Service 層？**
如果沒有 `Service` 層，**Controller 會直接調用 `Model`，導致控制器的邏輯變得混亂且難以測試**。  
加上 `Service` 層後，可以讓：
- **Controller 只專注於處理 HTTP Request/Response**。
- **Service 層負責業務邏輯，可以更容易進行單元測試**。
- **Model 只專注於資料存取，不包含額外邏輯**，保持關注點分離。

---

### **🔥 總結**
- **Model** 👉 對應 **資料庫**，負責 **CRUD 操作**。
- **Service** 👉 負責 **業務邏輯**，封裝操作，讓程式碼更可讀、可測試。
- **Controller** 👉 負責 **接收 HTTP 請求**，調用 `Service`，並回應結果給 `Client`。

這種結構讓 **程式更清晰、可維護、可測試**，是 Express.js 中的 **最佳實踐** 🚀。
