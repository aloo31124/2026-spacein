  getUserRole(userId: string) {
    const userToRoleTable = this.fireStoreService.fireStoreTabelNameList.UserToRole;
    return this.fireStoreService.getDataByField(userToRoleTable, "userId", userId);
  }

  getDataByField(tableName: string, fieldName: string, fieldValue: string):Promise<any> {
    const q = query(collection(this.firestore, tableName), where(fieldName, "==", fieldValue));
    return getDocs(q);
  }
