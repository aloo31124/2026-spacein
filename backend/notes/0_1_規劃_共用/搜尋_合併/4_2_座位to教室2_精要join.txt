static async joinRoomByRoomId(roomToSiteList) {
    const roomMap = new Map((await Room.getAllRoomList()).map(room => [room.id, room]));

    return roomToSiteList.map(site => ({
        ...site,
        roomName: roomMap.get(site.roomId)?.name || "未知教室",
        roomOwner: roomMap.get(site.roomId)?.ownerId || "未知擁有者",
    }));
}
